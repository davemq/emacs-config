** Org random scatter for deadlines

#+begin_src emacs-lisp :tangle yes :comments both
;; org agenda bulk action for random scatter for deadlines
(defun drm:org-agenda-deadline ()
  (unless (org-agenda-check-type nil 'agenda 'todo)
    (user-error "Can't scatter tasks in \"%s\" agenda view" org-agenda-type))
  (let ((days (read-number "Scatter tasks across how many days: " 7)))
    (let ((distance (1+ (random days))))
      ;; Silently fail when try to replan a sexp entry.
      (ignore-errors
	(let* ((date (calendar-gregorian-from-absolute
		      (+ (org-today) distance)))
	       (time (encode-time 0 0 0 (nth 1 date) (nth 0 date)
				  (nth 2 date))))
	  (org-agenda-deadline nil time)
	  )
	)
      )
    )
  )
#+end_src

(use-package gnus
  :ensure t
  :hook (gnus-setup-news . gnus-demon-init)
  :init
  (defun gnus-user-format-function-t (headers)
    "Return tag from subject"
    (let*
	((subject (aref headers 1))
	 (cb (string-match "\\[[^]]+]" subject)))
      (if cb
	  (substring subject 0 (match-end 0))
	"")
      )
    )
  (defun gnus-user-format-function-s (headers)
    "Return subject - tag"
    (let*
	((subject (aref headers 1))
	 (cb (string-match "\\[[^]]+]" subject)))
      (string-trim-left 
       (if cb
	   (substring subject (min (match-end 0) (length subject)))
	 subject
	 )
       )
      )
    )
  (defun gnus-user-format-function-b (headers)
    "Return subject if not blank otherwise first 140 characters of body"
    (let
	((subject (aref headers 1))
	 (article (aref headers 0)))
      (if (or (zerop (length subject))
	      (string= subject "(untitled)"))
	  ;; get body
	  (if gnus-newsgroup-name
	      (with-temp-buffer
		(gnus-request-article-this-buffer article gnus-newsgroup-name)
		(gnus-narrow-to-body)
		(string-replace "\n" "\\n"
				(replace-regexp-in-string "<[^>]*>" ""
							  (buffer-substring-no-properties
							   (point) (+ (point) 140)
							   )
							  )
				)
		)
	    subject
	    )
	subject)
      )
    )
  :custom
  (gnus-adaptive-pretty-print t)
  (gnus-decay-scores t)

     (gnus-catchup-mark (subject -10) (followup -10) (from -5))
     (gnus-killed-mark (from -4) (subject -20))
     (gnus-del-mark (from -4) (subject -15))))
  (gnus-demon-handlers '((gnus-demon-scan-news 13 1)))
  (gnus-group-mode-hook '(gnus-agent-mode gnus-topic-mode))
